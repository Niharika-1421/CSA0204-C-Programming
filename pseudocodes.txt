1. STRING LENGTH
STRING_LENGTH

READ STRING
SET LENGTH = 0
WHILE CHARACTER IS NOT END OF STRING
    LENGTH = LENGTH + 1
END WHILE
PRINT LENGTH

END

2. ALPHABETICAL ORDER OF NAMES
ALPHABETICAL_ORDER

READ N
READ N NAMES
FOR I = 0 TO N-2
    FOR J = I+1 TO N-1
        IF NAME[I] > NAME[J]
            SWAP NAME[I] AND NAME[J]
        END IF
    END FOR
END FOR
PRINT NAMES

END

3. PALINDROME
PALINDROME

READ STRING
SET FLAG = TRUE
COMPARE FIRST AND LAST CHARACTERS
IF ALL CHARACTERS MATCH
    PRINT PALINDROME
ELSE
    PRINT NOT PALINDROME
END IF

END

4. COUNT VOWELS, CONSONANTS, SPECIAL CHARACTERS
COUNT_CHARACTERS

READ STRING
SET VOWELS = 0
SET CONSONANTS = 0
SET SPECIAL = 0

FOR EACH CHARACTER
    IF VOWEL
        VOWELS++
    ELSE IF ALPHABET
        CONSONANTS++
    ELSE
        SPECIAL++
    END IF
END FOR

PRINT VOWELS
PRINT CONSONANTS
PRINT SPECIAL

END

5. LOWERCASE TO UPPERCASE
LOWERCASE_TO_UPPERCASE

READ STRING
FOR EACH CHARACTER
    IF LOWERCASE
        CONVERT TO UPPERCASE
    END IF
END FOR
PRINT STRING

END

6. LARGEST OF THREE NUMBERS
LARGEST_OF_THREE

READ A, B, C

IF A>B AND A>C
    PRINT A
ELSE IF B>C
    PRINT B
ELSE
    PRINT C
END IF

END

7. ARMSTRONG NUMBER
ARMSTRONG

READ NUMBER
COPY NUMBER TO TEMP
SET SUM = 0

WHILE TEMP > 0
    FIND LAST DIGIT
    ADD CUBE OF DIGIT TO SUM
    REMOVE LAST DIGIT
END WHILE

IF SUM = NUMBER
    PRINT ARMSTRONG
ELSE
    PRINT NOT ARMSTRONG
END IF

END

8. SUM OF DIGITS
SUM_OF_DIGITS

READ NUMBER
SET SUM = 0

WHILE NUMBER > 0
    FIND LAST DIGIT
    ADD TO SUM
    REMOVE LAST DIGIT
END WHILE

PRINT SUM

END

17. POSITIVE AND NEGATIVE COUNT
POSITIVE_NEGATIVE_COUNT

READ ARRAY

SET POS=0
SET NEG=0

FOR EACH ELEMENT
    IF ELEMENT>0
        POS++
    ELSE IF ELEMENT<0
        NEG++
    END IF
END FOR

PRINT POS
PRINT NEG

END

END

20. SORT IN INCREASING ORDER
SORT_ARRAY

READ ARRAY

FOR I=1 TO N-1
    FOR J=I+1 TO N
        IF ARRAY[I]>ARRAY[J]
            SWAP
        END IF
    END FOR
END FOR

PRINT ARRAY

END

28. MATRIX MULTIPLICATION
